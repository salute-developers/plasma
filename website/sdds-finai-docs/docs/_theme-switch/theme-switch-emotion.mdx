```jsx
import React, { useState } from 'react';
import { Global, css } from '@emotion/react';

import { Switch } from '@salutejs/sdds-finai/emotion';
import { sdds_finai__light, sdds_finai__dark } from '@salutejs/sdds-themes';

const lightThemeStyle = css(sdds_finai__light);
const darkThemeStyle = css(sdds_finai__dark);

const App = () => {
    const [theme, setTheme] = useState('light');

    return (
        <>
            <Global styles={theme === 'light' ? lightThemeStyle : darkThemeStyle} />

            <div className="wrapper">
                <Switch
                    label={`app theme: ${theme}`}
                    onChange={() => setTheme((theme) => (theme === 'light' ? 'dark' : 'light'))}
                />
            </div>
        </>
    );
};

export default App;
```

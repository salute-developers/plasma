:::info
Если в качестве основы web-приложения вы используете [Next.js](https://nextjs.org/), добавьте ссылки на шрифты через `next/head` — например, в `./pages/index.tsx`.
:::

```tsx
import Head from "next/head";

import { H2, Button } from "@salutejs/sdds-cs";

export default function Home() {
  return (
    <>
      <Head>
        <title>Create Next App with sdds-cs components</title>
        <link rel="stylesheet" href="https://cdn-app.sberdevices.ru/shared-static/0.0.0/styles/SBSansText.0.2.0.css" />
        <link
            rel="stylesheet"
            href="https://cdn-app.sberdevices.ru/shared-static/0.0.0/styles/SBSansDisplay.0.2.0.css"
        />
        <link
            rel="stylesheet"
            href="https://cdn-app.sberdevices.ru/shared-static/0.0.0/styles/SBSansTextMono.0.2.0.css"
        />
      </Head>
      <div>
        <main>
            <div>
                <H2> SDDS CS </H2>
                <Button text="Hello" />
            </div>
        </main>
      </div>
    </>
  );
}
```

:::info
Если в качестве основы web-приложения вы используете [Next.js](https://nextjs.org/), добавьте ссылки на шрифты через `next/head` — например, в `./pages/index.tsx`.
:::

```tsx
import Head from "next/head";

import { H2, Button } from "@salutejs/sdds-sbcom";

export default function Home() {
  return (
    <>
      <Head>
        <title>Create Next App with sdds-sbcom components</title>
        <link
            href="https://fonts.googleapis.com/css2?family=Roboto:ital,wght@0,400..700;1,400..700&display=swap"
            rel="stylesheet"
        />
        <link
            rel="stylesheet"
            href="https://cdn-app.sberdevices.ru/shared-static/0.0.0/styles/SBSansTextMono.0.2.0.css"
        />
      </Head>
      <div>
        <main>
            <div>
                <H2> SDDS SBCOM </H2>
                <Button text="Hello" />
            </div>
        </main>
      </div>
    </>
  );
}
```
